# Відсутня обов'язкова опція -> повідомлення про помилку, код виходу 1
node main.js -h 127.0.0.1 -p 3000; echo "exit=$?"

# Некоректний порт -> повідомлення про помилку, код виходу 1
node main.js -h 127.0.0.1 -p abc -c ./cache; echo "exit=$?"

curl -i http://127.0.0.1:3000/abc

curl -i http://127.0.0.1:3000/200
curl -i -X POST http://127.0.0.1:3000/200

=== метод GET ===

# Картинки немає в кеші -> 200, береться з http.cat (X-Image-Source: http.cat)
curl -s -o /dev/null -D - -w 'час: %{time_total} с\n' http://127.0.0.1:3000/418

# Повторний GET тієї самої картинки -> 200 з кешу (X-Image-Source: cache), помітно швидше
curl -s -o /dev/null -D - -w 'час: %{time_total} с\n' http://127.0.0.1:3000/418

# Картинка збереглася в теці кешу
ls -l cache

# Коду немає на http.cat -> 404, у кеш нічого не потрапляє
curl -i http://127.0.0.1:3000/999
ls -l cache

# PUT ще не реалізовано -> 405, Allow: GET
curl -i -X PUT http://127.0.0.1:3000/200

=== метод PUT ===

# Власна картинка (будь-який .jpg) -> 201
curl -i -X PUT --data-binary @my-cat.jpg http://127.0.0.1:3000/200

# GET після PUT -> 200 з кешу, повертається саме власна картинка
curl -s -o /tmp/got.jpg -D - http://127.0.0.1:3000/200
cmp my-cat.jpg /tmp/got.jpg && echo "однакові"

# PUT без тіла -> 400
curl -i -X PUT http://127.0.0.1:3000/201

# Тіло більше за 10 МБ -> 413
head -c 11000000 /dev/zero > /tmp/big.bin
curl -i -X PUT --data-binary @/tmp/big.bin http://127.0.0.1:3000/202

# DELETE ще не реалізовано -> 405, Allow: GET, PUT
curl -i -X DELETE http://127.0.0.1:3000/200


=== метод DELETE ===

# Картинка є в кеші -> 200
curl -i -X DELETE http://127.0.0.1:3000/200

# Картинки вже немає -> 404
curl -i -X DELETE http://127.0.0.1:3000/200

# Недозволений метод -> 405, Allow: GET, PUT, DELETE
curl -i -X POST http://127.0.0.1:3000/200
